// app/api/auth/apple/route.ts
// Apple sign-in / sign-up. Apple has confirmed the person owns the email (verifyAppleToken
// refuses unverified emails), so, the same as Google:
//  - a new email gets a confirmed account
//  - an existing account is signed in; if its email was never confirmed, it's confirmed now and its
//    password is cleared, because whoever set that password never proved they own the email
//  - banned and deleted accounts can't sign in; suspended accounts can
import { NextRequest, NextResponse } from "next/server";
import { db } from "@/app/lib/servers/db";
import {
    verifyAppleToken,
    generateAccessToken,
    generateRefreshToken,
    isAuthorRole,
} from "@/app/lib/servers/auth";
import { generateUniquePublicId } from "@/app/lib/servers/publicId";
import { withTransaction } from "@/app/lib/servers/withTransaction";
import { cancelCodes } from "@/app/lib/servers/emailCode";
import { CLOSED_MESSAGE, isClosedAccount } from "@/app/lib/servers/loginLimit";
import { sendWelcomeEmail } from "@/app/lib/servers/email/welcomeEmail";

const PROFILE_INSERT = `
    INSERT INTO user_profiles (
        user_id, public_id, full_name, pen_name, gender, email, dob, nationality, country,
        id_number, id_validity, id_type, phone_number, phone_country, profile_image,
        id_document_front, id_document_back, e_signature, id_expiration_date
    )
    VALUES ($1, $2, $3, '', '', $4, NULL, '', '', '', '', '', '', '', NULL, NULL, NULL, NULL, NULL)`;

export async function POST(request: NextRequest) {
    try {
        let body;
        try {
            body = await request.json();
        } catch {
            return NextResponse.json({ error: "Invalid request body" }, { status: 400 });
        }

        const idToken = body?.idToken;
        if (!idToken || typeof idToken !== "string") {
            return NextResponse.json({ error: "Missing Apple ID token" }, { status: 400 });
        }

        // Apple sends the name only on the first sign-in, from the browser; it's only used as a display name
        const fullName = typeof body?.fullName === "string" ? body.fullName.trim().slice(0, 100) : "";

        let appleUser;
        try {
            appleUser = await verifyAppleToken(idToken);
        } catch (error) {
            console.error("Apple token check failed:", error);
            const unverified = error instanceof Error && error.message === "Apple email is not verified";
            return NextResponse.json(
                { error: unverified ? "Your Apple email isn't verified. Sign up with email and password instead." : "Failed to verify Apple sign-in" },
                { status: 401 }
            );
        }

        const normalizedEmail = appleUser.email.trim().toLowerCase();

        const existing = await db.query(
            `SELECT id, email, role, email_verified_at, account_status FROM users WHERE email = $1`,
            [normalizedEmail]
        );

        let user;

        if (existing.rows.length > 0) {
            // Existing user
            user = existing.rows[0];

            if (!isAuthorRole(user.role)) {
                return NextResponse.json(
                    { error: "This account doesn't have author access.", code: "NOT_AUTHOR" },
                    { status: 403 }
                );
            }

            if (isClosedAccount(user.account_status)) {
                return NextResponse.json({ error: CLOSED_MESSAGE, code: "ACCOUNT_CLOSED" }, { status: 403 });
            }

            // Never confirmed: confirm it now and clear the password nobody proved they own
            if (!user.email_verified_at) {
                await db.query(
                    `UPDATE users SET email_verified_at = now(), password_hash = NULL WHERE id = $1 AND email_verified_at IS NULL`,
                    [user.id]
                );
                await cancelCodes(user.id, "verify_email");
            }

            // Create missing profile
            const profileCheck = await db.query(
                `SELECT id FROM user_profiles WHERE user_id = $1`,
                [user.id]
            );
            if (profileCheck.rows.length === 0) {
                await db.query(PROFILE_INSERT, [user.id, generateUniquePublicId(), fullName, user.email]);
            }
        } else {
            // New user → confirmed user + profile together
            user = await withTransaction(async (client) => {
                const { rows } = await client.query(
                    `INSERT INTO users (email, password_hash, role, email_verified_at)
                     VALUES ($1, NULL, $2, now())
                     RETURNING id, email, role`,
                    [normalizedEmail, "author"]
                );
                const newUser = rows[0];
                await client.query(PROFILE_INSERT, [newUser.id, generateUniquePublicId(), fullName, newUser.email]);
                return newUser;
            });

            try {
                await sendWelcomeEmail({
                    email: user.email,
                    firstName: fullName.split(/\s+/)[0] || undefined,
                });
            } catch (emailError) {
                console.error("Welcome email error:", emailError);
            }
        }

        const accessToken = await generateAccessToken({ id: user.id, email: user.email, role: user.role });
        const refreshToken = await generateRefreshToken({ id: user.id, email: user.email, role: user.role });

        const response = NextResponse.json({ id: user.id, email: user.email });

        const baseCookieOptions = {
            httpOnly: true,
            secure: process.env.NODE_ENV === "production",
            sameSite: "lax" as const,
            path: "/",
        };

        response.cookies.set("access_token", accessToken, { ...baseCookieOptions, maxAge: 30 * 60 });
        response.cookies.set("refresh_token", refreshToken, { ...baseCookieOptions, maxAge: 30 * 24 * 60 * 60 });

        return response;
    } catch (error) {
        console.error("Apple authentication error:", error);
        return NextResponse.json(
            { error: "Something went wrong while signing in with Apple" },
            { status: 500 }
        );
    }
}
